-- Head-to-head comparison
SELECT
  CASE WHEN created_at >= '2026-01-31 06:00:00+00' THEN 'A_GROK_FIRST' ELSE 'B_OLD_ENSEMBLE' END as engine,
  COUNT(*) as evaluated,
