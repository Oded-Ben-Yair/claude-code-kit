## Song Data Structure (for 2-column layout)

Each song in `songs-parsed.json` has:
```typescript
interface Song {
  number: number;
  titleEnglish: string;
  titleHebrew?: string;
  subtitle: string;
  key: string;
  capo: number;
  style?: string;
  category: 'hebrew' | 'international';
  chordsUsed: string[];
  verses: SongVerse[];  // <- This is what needs 2-column layout
  notes?: string;
}

interface SongVerse {
  label: string;  // "Verse 1", "Chorus", "Bridge", etc.
  lines: SongLine[];
}

interface SongLine {
  text: string;
  chords?: string;
  transliteration?: string;  // For international songs
}
```
